إعداد النشر لصاحب WebForge
  1. سجّل تطبيق Web بمشروع Firebase المركزي، فعّل طرق الدخول Email/Password وGoogle وFacebook، وضيف نطاق Vercel ضمن Authorized domains.
  2. لـFacebook، سجّل تطبيق Meta وأضف OAuth redirect URI اللي بيظهر بإعدادات Firebase إلى Valid OAuth Redirect URIs في Meta.
  3. أنشئ Cloud Firestore. بمشروع مركزي جديد، انشر القواعد أدناه. إذا عندك قواعد قائمة، ادمج قواعد WebForge معها.
  4. بملف index.html، حطّ إعدادات تطبيق Web بصيغة JSON داخل webforge-cloud-config وبعدين أعد الرفع.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owner(uid) { return signedIn() && request.auth.uid == uid; }
    function validSlug(slug) {
      return slug is string && slug.size() >= 3 && slug.size() <= 48
        && slug.matches('^[a-z0-9]+(-[a-z0-9]+)*$');
    }
    function validProject(d) {
      return d.keys().hasAll(['draft','slug','revision','updatedAt'])
        && d.keys().hasOnly(['draft','slug','revision','updatedAt'])
        && d.draft is map && d.revision is int && d.revision > 0
        && (d.slug == null || validSlug(d.slug))
        && d.updatedAt == request.time;
    }
    function publishedTo(uid, id, slug) {
      let path = /databases/$(database)/documents/webforgePublished/$(slug);
      return existsAfter(path) && getAfter(path).data.uid == uid
        && getAfter(path).data.projectId == id;
    }
    match /webforgeUsers/{uid}/projects/{id} {
      allow get, list: if owner(uid);
      allow create: if owner(uid) && validProject(request.resource.data)
        && request.resource.data.revision == 1
        && (request.resource.data.slug == null || publishedTo(uid, id, request.resource.data.slug));
      allow update: if owner(uid) && validProject(request.resource.data)
        && request.resource.data.revision == resource.data.revision + 1
        && (request.resource.data.slug == resource.data.slug
          || (resource.data.slug == null && validSlug(request.resource.data.slug)
            && publishedTo(uid, id, request.resource.data.slug)));
    }
    match /webforgePublished/{slug} {
      function validPublication(d) {
        let path = /databases/$(database)/documents/webforgeUsers/$(d.uid)/projects/$(d.projectId);
        return d.keys().hasAll(['uid','projectId','site','updatedAt'])
          && d.keys().hasOnly(['uid','projectId','site','updatedAt'])
          && owner(d.uid) && d.projectId is string && validSlug(slug)
          && d.site is map && d.updatedAt == request.time
          && existsAfter(path) && getAfter(path).data.slug == slug
          && getAfter(path).data.draft == d.site;
      }
      allow get: if true;
      allow create: if validPublication(request.resource.data);
      allow update: if owner(resource.data.uid)
        && request.resource.data.uid == resource.data.uid
        && request.resource.data.projectId == resource.data.projectId
        && validPublication(request.resource.data);
    }
  }
}

ربط Firebase اختياري

احفظ مسودّة واحدة بحساب Google في مشروع Firebase الخاص فيك، واسترجعها من جهاز تاني.

  1. بـFirebase Console، افتح مشروعك أو اعمل مشروع وسجّل تطبيق Web.
  2. فعّل Authentication → Google، وضيف اسم نطاق WebForge المستضاف ضمن Authorized domains.
  3. أنشئ Cloud Firestore، وضيف قاعدة المسودّات التالية ضمن match /databases/{database}/documents. حافظ على قواعد تطبيقاتك الموجودة.
match /webforgeDrafts/{uid} {
  allow read, write: if request.auth != null
    && request.auth.uid == uid;
}

استعمل إعدادات تطبيق Web فقط. اكتب أسماء الحقول والقيم بين علامتي اقتباس مزدوجتين.
